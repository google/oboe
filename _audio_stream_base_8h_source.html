416 virtual Result isValidConfig() {
417 switch (mFormat) {
418 case AudioFormat::Unspecified:
419 case AudioFormat::I16:
420 case AudioFormat::Float:
421 case AudioFormat::I24:
422 case AudioFormat::I32:
423 case AudioFormat::IEC61937:
424 case AudioFormat::MP3:
425 case AudioFormat::AAC_LC:
426 case AudioFormat::AAC_HE_V1:
427 case AudioFormat::AAC_HE_V2:
428 case AudioFormat::AAC_ELD:
429 case AudioFormat::AAC_XHE:
430 case AudioFormat::OPUS:
431 break;
432
433 default:
434 return Result::ErrorInvalidFormat;
435 }
436
437 if (mChannelAcnMask != ChannelAcnMask::Unspecified
438 && !isValidChannelAcnMask(mChannelAcnMask)) {
439 return Result::ErrorIllegalArgument;
440 }
441
442 switch (mSampleRateConversionQuality) {
443 case SampleRateConversionQuality::None:
444 case SampleRateConversionQuality::Fastest:
445 case SampleRateConversionQuality::Low:
446 case SampleRateConversionQuality::Medium:
447 case SampleRateConversionQuality::High:
448 case SampleRateConversionQuality::Best:
449 return Result::OK;
450 default:
451 return Result::ErrorIllegalArgument;
452 }
453 }
